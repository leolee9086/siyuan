# 示例：每分钟导出最新文档

这是一个定时任务示例，演示如何每分钟检查最近修改的文档，并将其 Markdown 内容导出到指定文件夹。

## 任务配置

```go
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"time"
	"siyuan/siyuan"
)

// 任务名称
var Name = "导出最新文档"

// 任务描述
var Description = "每分钟检查并导出最近更新的文档内容到指定目录"

// Cron 表达式：每分钟执行一次
var Schedule = "*/1 * * * *"

// 导出目录：请修改为你实际存在的目录
var ExportDir = "d:/tmp/siyuan_latest_export"

// Run 任务入口函数
func Run(ctx *siyuan.Context) error {
	// 1. 查找最新更新的文档 (非虚拟块)
	// 使用 SQL 查询 API
	stmt := "SELECT * FROM blocks WHERE type = 'd' ORDER BY updated DESC LIMIT 1"
	ret, err := ctx.Call("/api/query/sql", map[string]interface{}{
		"stmt": stmt,
	})
	if err != nil {
		siyuan.日志错误(fmt.Sprintf("查询文档失败: %v", err))
		return err
	}

	// 解析查询结果
	// 注意：对于返回列表的 API (如 SQL)，结果会被包装在 result 字段中
	dataList, ok := ret["result"].([]interface{})
	if !ok || len(dataList) == 0 {
		siyuan.日志信息("没有找到任何文档")
		return nil
	}

	// 获取第一行数据
	row := dataList[0].(map[string]interface{})
	docID := row["id"].(string)
	docContent := row["content"].(string) // 文档标题通常存储在 content 字段
	
	siyuan.日志信息(fmt.Sprintf("发现最新文档: %s (%s)", docContent, docID))

	// 2. 导出 Markdown 内容
	// 使用 exportMdContent API
	retExport, err := ctx.Call("/api/export/exportMdContent", map[string]interface{}{
		"id": docID,
	})
	if err != nil {
		return fmt.Errorf("调用导出 API 失败: %v", err)
	}

	// 获取 Markdown 内容
	// exportMdContent API 直接返回包含 content 和 hpath 的 map
	var mdContent string
	
	if c, ok := retExport["content"].(string); ok {
		mdContent = c
	}

	if mdContent == "" {
		return fmt.Errorf("无法获取文档内容 (可能是空文档或 API 响应格式不匹配)")
	}

	// 3. 写入文件
	if err := os.MkdirAll(ExportDir, 0755); err != nil {
		return fmt.Errorf("创建目录失败: %v", err)
	}

	// 生成文件名：标题_时间.md
	// 简单过滤非法字符
	safeTitle := filepath.Base(docContent) 
	if safeTitle == "" || safeTitle == "." {
		safeTitle = "Untitled"
	}
	
	timestamp := time.Now().Format("20060102_150405")
	filename := fmt.Sprintf("%s_%s.md", safeTitle, timestamp)
	fullPath := filepath.Join(ExportDir, filename)

	if err := os.WriteFile(fullPath, []byte(mdContent), 0644); err != nil {
		return fmt.Errorf("写入文件失败: %v", err)
	}

	siyuan.日志信息("成功导出到: " + fullPath)
	return nil
}
```

## 使用说明

1.  **设置属性**：
    *   `ext-lang`: `go`
    *   `ext-type`: `cronjob`
2.  **修改配置**：
    *   修改 `ExportDir` 变量为你希望保存文件的目录路径。
3.  **编译运行**：
    *   在插件管理界面或侧边栏点击编译/注册。
    *   启用任务。
4.  **验证**：
    *   修改任意一篇笔记。
    *   等待一分钟（或手动触发）。
    *   检查目标目录下是否生成了对应的 Markdown 文件。
