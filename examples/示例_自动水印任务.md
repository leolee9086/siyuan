# 自动水印任务

这是一个文学编程风格的定时任务脚本。它会监听指定文件夹，为新增的图片自动添加水印。

## 任务配置

首先定义任务的基本配置信息：

```go
package main

import (
	"fmt"
	"image"
	"image/color"
	"image/draw"
	_ "image/jpeg"
	"image/png"
	"io/ioutil"
	"os"
	"path/filepath"
	"strings"

	"siyuan/siyuan"
)

// 任务名称
var Name = "自动添加图片水印 (UTF-8支持)"

// 任务描述
var Description = "监听指定文件夹，使用系统字体为新图片添加水印，支持中文"

// 调度表达式（每5分钟执行一次）
var Schedule = "*/5 * * * *"

// 监听的目录路径
var WatchDir = "D:/Pictures/待处理"

// 全局字体缓存
var cachedFace interface{}

// Run 任务主函数
func Run(ctx *siyuan.Context) error {
	siyuan.日志信息("开始执行 UTF-8 水印任务")

	// 1. 获取水印配置
	config := siyuan.获取图片水印配置()
	if config == nil || config.ImageWatermarkStr == "" {
		siyuan.日志警告("水印配置为空，跳过")
		return nil
	}
	watermarkText := config.ImageWatermarkStr

	// 2. 加载字体 (只加载一次)
	if cachedFace == nil {
		face, err := loadChineseFont(ctx)
		if err != nil {
			siyuan.日志错误("加载字体失败: " + err.Error())
			return err
		}
		cachedFace = face
	}

	// 3. 扫描并处理图片
	files := scanImages(WatchDir)
	for _, file := range files {
		if isProcessed(file) {
			continue
		}

		siyuan.日志信息("正在处理: " + file)
		output := generateOutputPath(file)

		if err := addWatermark(file, output, watermarkText, cachedFace); err != nil {
			siyuan.日志错误("处理失败 [" + file + "]: " + err.Error())
		} else {
			siyuan.日志信息("处理成功: " + output)
		}
	}
	return nil
}

// loadChineseFont 加载支持中文的字体
func loadChineseFont(ctx *siyuan.Context) (interface{}, error) {
	// 略... 保持原样 (调用 ctx.Call 获取字体 或 硬编码)
	// 此处省略 loadChineseFont 内部逻辑，假设与之前一致，只展示关键 diff
	
	// 简单示例实现：
	fontPaths := []string{
		"C:/Windows/Fonts/msyh.ttc",
		"C:/Windows/Fonts/simhei.ttf",
	}
	var fontData []byte
	for _, p := range fontPaths {
		data, err := ioutil.ReadFile(p)
		if err == nil {
			fontData = data
			siyuan.日志信息("已加载字体: " + p)
			break
		}
	}
	if fontData == nil {
		return nil, fmt.Errorf("找不到合适的中文字体")
	}
	// 使用内核 Helper
	return siyuan.LoadOpenTypeFont(fontData, 24)
}

// addWatermark 添加文字水印
func addWatermark(srcPath, dstPath, text string, face interface{}) error {
	// 读取原图
	f, err := os.Open(srcPath)
	if err != nil {
		return err
	}
	defer f.Close()

	srcImg, _, err := image.Decode(f)
	if err != nil {
		return err
	}

	bounds := srcImg.Bounds()
	width := bounds.Dx()
	height := bounds.Dy()

	// 创建可绘图的 RGBA 图像
	rgba := image.NewRGBA(bounds)
	draw.Draw(rgba, bounds, srcImg, bounds.Min, draw.Src)

	// 计算文本尺寸
	textW, textH, _ := siyuan.MeasureText(face, text)
	
	// 计算间距 (参考 kernel/cronjob/watermark.go)
	spacingX := textW * 3
	if spacingX < 150 {
		spacingX = 150
	}
	spacingY := textH * 8
	if spacingY < 100 {
		spacingY = 100
	}

	// 旋转参数 (-45度)
	angle := -45.0
	radian := angle * 3.14159265358979323846 / 180.0
	cos := 0.70710678118 // math.Cos(-45 deg)
	sin := -0.70710678118 // math.Sin(-45 deg)
	// 在脚本里可能没有 math 包，这里直接硬编码 -45 度

	// 水印颜色：半透明灰色 (参考 kernel)
	// R: 128, G: 128, B: 128, A: 100
	c := color.RGBA{128, 128, 128, 100}

	siyuan.日志信息(fmt.Sprintf("开始绘制平铺水印: %dx%d, 文本: %dx%d, 间距: %dx%d", 
		width, height, textW, textH, spacingX, spacingY))

	// 覆盖式循环绘制
	// 范围扩大以覆盖旋转后的区域
	for y := -height; y < height*2; y += spacingY {
		for x := -width; x < width*2; x += spacingX {
			// 旋转坐标变换
			// X' = x*cos - y*sin
			// Y' = x*sin + y*cos
			rx := int(float64(x)*cos - float64(y)*sin)
			ry := int(float64(x)*sin + float64(y)*cos)

			// 检查边界 (宽松一点)
			if rx >= -100 && rx < width+100 && ry >= -50 && ry < height+50 {
				siyuan.DrawText(rgba, face, rx, ry, text, c)
			}
		}
	}

	// 保存结果
	outF, err := os.Create(dstPath)
	if err != nil {
		return err
	}
	defer outF.Close()

	return png.Encode(outF, rgba)
}
```

```go
// scanImages 扫描目录中的图片文件
func scanImages(dir string) []string {
	var files []string
	filepath.Walk(dir, func(path string, info os.FileInfo, err error) error {
		if err == nil && !info.IsDir() {
			ext := strings.ToLower(filepath.Ext(path))
			if ext == ".jpg" || ext == ".png" {
				files = append(files, path)
			}
		}
		return nil
	})
	return files
}

func isProcessed(path string) bool {
	return strings.Contains(path, "_watermarked")
}

func generateOutputPath(input string) string {
	ext := filepath.Ext(input)
	base := strings.TrimSuffix(input, ext)
	return base + "_watermarked.png" // 强制存为 PNG 以保持质量
}
```

## 使用说明

1. 将此文档的 `ext-lang` 属性设为 `go`
2. 将 `ext-type` 属性设为 `cronjob`
3. 修改 `WatchDir` 为你要监听的实际目录
4. 在思源的导出设置中配置水印文本
5. 通过 API 或界面启用此任务

任务启用后会每5分钟检查一次目录，为所有未处理的图片添加水印。
